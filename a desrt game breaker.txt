local function weld(weldto)
    local weldtarget

    for _,obj in pairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") or obj:IsA("Model") then
            local welds = 0
            local foundgyro = false
            local foundpos = false
            local foundweld

            for _,v in pairs(obj:GetChildren()) do
                if v:IsA("WeldConstraint") and v.Name:lower():find("objectweld") then
                    welds += 1
                    foundweld = v
                elseif v:IsA("BodyGyro") then
                    foundgyro = true
                elseif v:IsA("BodyPosition") then
                    foundpos = true
                end
            end

            if welds >= 2 and foundgyro and foundpos and foundweld then
                if foundweld.Part1 then
                    weldtarget = foundweld.Part1
                    break
                end
            end
        end
    end

    if not weldtarget then
        return
    end

    keypress(46)
    task.wait(0.1)
    keyrelease(46)

    weldinvoke:InvokeServer(unpack({
        weldtarget,
        weldtarget.Parent,
        {
            {
                obj = weldto,
                n = 1
            }
        }
    }))
end

local function GetClosestCharacter()
	local closest, dist = nil, math.huge
	local root = LocalPlayer.Character.HumanoidRootPart

	for _,v in pairs(Players:GetPlayers()) do
		if v ~= LocalPlayer and v.Character and v.Character:FindFirstChild("HumanoidRootPart") then
			local mag = (root.Position - v.Character.HumanoidRootPart.Position).Magnitude

			if mag < dist then
				dist = mag
				closest = v.Character
			end
		end
	end

	return closest
end

local function GetClosestCar()
	local closest, dist = nil, math.huge
	local root = LocalPlayer.Character.HumanoidRootPart

	for _,v in pairs(workspace:GetDescendants()) do
		if v:IsA("Model") and v.Name:lower():find("car") and v.PrimaryPart then
			local mag = (root.Position - v.PrimaryPart.Position).Magnitude

			if mag < dist then
				dist = mag
				closest = v
			end
		end
	end

	return closest
end

local toggledstuff = {}
local enabledtoggles = {}

local function refreshrvalues()
    holding = LocalPlayer.PlayerGui.main.handtarget1
    twohands = LocalPlayer.PlayerGui.main.twohands

    health = LocalPlayer.PlayerGui.main.health
    hunger = LocalPlayer.PlayerGui.main.hunger
    stamina = LocalPlayer.PlayerGui.main.stamina
end

function toggle(value, toggleenabled)
    if toggleenabled then
        if toggledstuff[value] then
            return
        end

        toggledstuff[value] = value.Changed:Connect(function()
            value.Value = 100
        end)

        value.Value = 100
    else
        if toggledstuff[value] then
            toggledstuff[value]:Disconnect()
            toggledstuff[value] = nil
        end
    end
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(1)

    refreshrvalues()

    if enabledtoggles["health"] then
        toggle(health, true)
    end

    if enabledtoggles["hunger"] then
        toggle(hunger, true)
    end

    if enabledtoggles["stamina"] then
        toggle(stamina, true)
    end

    if enabledtoggles["run"] then
        toggle(run, true)
    end
end)

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local damageRemote = ReplicatedStorage:WaitForChild("damage")

local damageme = false
local damageAmount = 0

local function fireDamage()
	local targets = {}

	for _, player in ipairs(Players:GetPlayers()) do
		if player and player.Parent == Players then
			if not damageme and player == Players.LocalPlayer then
				continue
			end

			table.insert(targets, {
				samplayer = player,
				sila = damageAmount
			})
		end
	end

	if #targets > 0 then
		damageRemote:FireServer(targets)
	end
end
